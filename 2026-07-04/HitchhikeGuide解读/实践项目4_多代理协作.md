```python
# communication/message.py
from typing import Any, Dict, Optional, List
from pydantic import BaseModel, Field
from datetime import datetime
from enum import Enum


class MessageType(str, Enum):
    """消息类型"""
    TASK = "task"           # 任务消息
    RESULT = "result"       # 结果消息
    QUERY = "query"         # 查询消息
    RESPONSE = "response"   # 响应消息
    ERROR = "error"         # 错误消息
    STATUS = "status"       # 状态消息


class Message(BaseModel):
    """消息基类"""
    id: str = Field(default_factory=lambda: str(datetime.now().timestamp()))
    type: MessageType
    sender: str
    receiver: str
    content: Any
    metadata: Dict[str, Any] = Field(default_factory=dict)
    timestamp: datetime = Field(default_factory=datetime.now)
    
    def to_dict(self) -> Dict:
        """转换为字典"""
        return {
            "id": self.id,
            "type": self.type.value,
            "sender": self.sender,
            "receiver": self.receiver,
            "content": self.content,
            "metadata": self.metadata,
            "timestamp": self.timestamp.isoformat()
        }


class TaskMessage(Message):
    """任务消息"""
    type: MessageType = MessageType.TASK
    task_id: str = ""
    task_description: str = ""
    priority: int = 1  # 1-5, 5最高


class ResultMessage(Message):
    """结果消息"""
    type: MessageType = MessageType.RESULT
    task_id: str = ""
    success: bool = True
    result: Any = None
    error: Optional[str] = None


class QueryMessage(Message):
    """查询消息"""
    type: MessageType = MessageType.QUERY
    query: str = ""
    context: Dict[str, Any] = Field(default_factory=dict)


class ResponseMessage(Message):
    """响应消息"""
    type: MessageType = MessageType.RESPONSE
    query_id: str = ""
    response: Any = None
```

### 3.2 通信通道

```python
# communication/channel.py
from typing import Dict, List, Callable, Optional
from collections import defaultdict
from communication.message import Message, MessageType
import threading
import queue


class MessageChannel:
    """
    消息通道
    管理代理间通信
    """
    
    def __init__(self):
        """初始化消息通道"""
        # 订阅者映射：agent_id -> 消息类型 -> 回调函数
        self.subscribers: Dict[str, Dict[MessageType, List[Callable]]] = defaultdict(
            lambda: defaultdict(list)
        )
        
        # 消息队列：agent_id -> 消息队列
        self.message_queues: Dict[str, queue.Queue] = defaultdict(queue.Queue)
        
        # 消息历史
        self.history: List[Message] = []
        
        # 锁
        self.lock = threading.Lock()
    
    def subscribe(self, agent_id: str, message_type: MessageType, callback: Callable):
        """
        订阅消息
        
        参数:
            agent_id: 代理ID
            message_type: 消息类型
            callback: 回调函数
        """
        with self.lock:
            self.subscribers[agent_id][message_type].append(callback)
    
    def unsubscribe(self, agent_id: str, message_type: MessageType):
        """
        取消订阅
        
        参数:
            agent_id: 代理ID
            message_type: 消息类型
        """
        with self.lock:
            if agent_id in self.subscribers:
                if message_type in self.subscribers[agent_id]:
                    del self.subscribers[agent_id][message_type]
    
    def publish(self, message: Message):
        """
        发布消息
        
        参数:
            message: 消息对象
        """
        with self.lock:
            # 添加到历史
            self.history.append(message)
            
            # 添加到接收者的消息队列
            self.message_queues[message.receiver].put(message)
            
            # 通知订阅者
            receiver = message.receiver
            msg_type = message.type
            
            if receiver in self.subscribers and msg_type in self.subscribers[receiver]:
                for callback in self.subscribers[receiver][msg_type]:
                    try:
                        callback(message)
                    except Exception as e:
                        print(f"回调执行失败: {e}")
    
    def get_message(self, agent_id: str, timeout: float = 1.0) -> Optional[Message]:
        """
        获取消息（阻塞）
        
        参数:
            agent_id: 代理ID
            timeout: 超时时间
            
        返回:
            消息对象或None
        """
        try:
            return self.message_queues[agent_id].get(timeout=timeout)
        except queue.Empty:
            return None
    
    def get_messages(self, agent_id: str) -> List[Message]:
        """
        获取所有待处理消息
        
        参数:
            agent_id: 代理ID
            
        返回:
            消息列表
        """
        messages = []
        while not self.message_queues[agent_id].empty():
            try:
                messages.append(self.message_queues[agent_id].get_nowait())
            except queue.Empty:
                break
        return messages
    
    def get_history(self, agent_id: str = None) -> List[Message]:
        """
        获取消息历史
        
        参数:
            agent_id: 代理ID（可选）
            
        返回:
            消息列表
        """
        if agent_id:
            return [m for m in self.history if m.sender == agent_id or m.receiver == agent_id]
        return self.history
    
    def clear(self):
        """清除所有消息"""
        with self.lock:
            self.message_queues.clear()
            self.history.clear()


class DirectChannel:
    """
    直接通道
    代理间直接通信
    """
    
    def __init__(self):
        """初始化直接通道"""
        self.agents: Dict[str, 'BaseAgent'] = {}
    
    def register_agent(self, agent_id: str, agent: 'BaseAgent'):
        """
        注册代理
        
        参数:
            agent_id: 代理ID
            agent: 代理对象
        """
        self.agents[agent_id] = agent
    
    def send_message(self, message: Message):
        """
        发送消息
        
        参数:
            message: 消息对象
        """
        if message.receiver in self.agents:
            self.agents[message.receiver].receive_message(message)
        else:
            print(f"警告: 接收者 {message.receiver} 未注册")
```

### 3.3 基础代理类

```python
# agents/base_agent.py
from typing import Any, Dict, List, Optional, Callable
from abc import ABC, abstractmethod
from openai import OpenAI
from communication.message import Message, MessageType, TaskMessage, ResultMessage
from communication.channel import MessageChannel


class BaseAgent(ABC):
    """
    基础代理类
    所有代理的父类
    """
    
    def __init__(
        self,
        agent_id: str,
        name: str,
        description: str,
        model: str = "gpt-3.5-turbo",
        channel: MessageChannel = None
    ):
        """
        初始化代理
        
        参数:
            agent_id: 代理唯一ID
            name: 代理名称
            description: 代理描述
            model: 使用的LLM模型
            channel: 消息通道
        """
        self.agent_id = agent_id
        self.name = name
        self.description = description
        self.model = model
        self.channel = channel
        
        # LLM客户端
        self.client = OpenAI()
        
        # 系统提示
        self.system_prompt = self._build_system_prompt()
        
        # 状态
        self.status = "idle"
        self.current_task = None
        
        # 消息处理器
        self.message_handlers: Dict[MessageType, Callable] = {
            MessageType.TASK: self._handle_task,
            MessageType.QUERY: self._handle_query,
        }
        
        # 注册到通道
        if self.channel:
            self._register_to_channel()
    
    def _build_system_prompt(self) -> str:
        """构建系统提示"""
        return f"""你是一个名为"{self.name}"的AI代理。

角色描述：{self.description}

你应该：
1. 专注于你的专业领域
2. 提供高质量的输出
3. 与其他代理协作完成任务
4. 遇到问题时及时报告

请始终保持专业和高效。"""
    
    def _register_to_channel(self):
        """注册到消息通道"""
        if self.channel:
            for msg_type, handler in self.message_handlers.items():
                self.channel.subscribe(self.agent_id, msg_type, handler)
    
    def send_message(self, receiver: str, content: Any, msg_type: MessageType = MessageType.RESULT):
        """
        发送消息
        
        参数:
            receiver: 接收者ID
            content: 消息内容
            msg_type: 消息类型
        """
        message = Message(
            type=msg_type,
            sender=self.agent_id,
            receiver=receiver,
            content=content
        )
        
        if self.channel:
            self.channel.publish(message)
        
        return message
    
    def receive_message(self, message: Message):
        """
        接收消息
        
        参数:
            message: 消息对象
        """
        handler = self.message_handlers.get(message.type)
        if handler:
            handler(message)
        else:
            print(f"[{self.name}] 未处理的消息类型: {message.type}")
    
    def _handle_task(self, message: Message):
        """
        处理任务消息
        
        参数:
            message: 任务消息
        """
        print(f"[{self.name}] 收到任务: {message.content}")
        self.status = "working"
        self.current_task = message
        
        try:
            # 执行任务
            result = self.execute_task(message.content)
            
            # 发送结果
            self.send_message(
                receiver=message.sender,
                content=result,
                msg_type=MessageType.RESULT
            )
        except Exception as e:
            # 发送错误
            self.send_message(
                receiver=message.sender,
                content={"error": str(e)},
                msg_type=MessageType.ERROR
            )
        finally:
            self.status = "idle"
            self.current_task = None
    
    def _handle_query(self, message: Message):
        """
        处理查询消息
        
        参数:
            message: 查询消息
        """
        print(f"[{self.name}] 收到查询: {message.content}")
        
        try:
            # 处理查询
            response = self.process_query(message.content)
            
            # 发送响应
            self.send_message(
                receiver=message.sender,
                content=response,
                msg_type=MessageType.RESPONSE
            )
        except Exception as e:
            self.send_message(
                receiver=message.sender,
                content={"error": str(e)},
                msg_type=MessageType.ERROR
            )
    
    @abstractmethod
    def execute_task(self, task: Any) -> Any:
        """
        执行任务（子类实现）
        
        参数:
            task: 任务描述
            
        返回:
            任务结果
        """
        pass
    
    def process_query(self, query: Any) -> Any:
        """
        处理查询（可覆盖）
        
        参数:
            query: 查询内容
            
        返回:
            响应内容
        """
        # 默认使用LLM处理查询
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": self.system_prompt},
                {"role": "user", "content": str(query)}
            ],
            temperature=0.3,
            max_tokens=500
        )
        
        return response.choices[0].message.content
    
    def call_llm(self, prompt: str, temperature: float = 0.3, max_tokens: int = 1000) -> str:
        """
        调用LLM
        
        参数:
            prompt: 用户提示
            temperature: 温度参数
            max_tokens: 最大令牌数
            
        返回:
            LLM响应
        """
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": self.system_prompt},
                {"role": "user", "content": prompt}
            ],
            temperature=temperature,
            max_tokens=max_tokens
        )
        
        return response.choices[0].message.content
    
    def get_status(self) -> Dict:
        """获取代理状态"""
        return {
            "agent_id": self.agent_id,
            "name": self.name,
            "status": self.status,
            "current_task": self.current_task.content if self.current_task else None
        }
```

```python
# agents/research_agent.py
from typing import Any, Dict, List
from agents.base_agent import BaseAgent
from communication.channel import MessageChannel


class ResearchAgent(BaseAgent):
    """
    研究代理
    专门负责信息收集和研究
    """
    
    def __init__(
        self,
        agent_id: str = "researcher",
        name: str = "研究代理",
        model: str = "gpt-3.5-turbo",
        channel: MessageChannel = None
    ):
        super().__init__(
            agent_id=agent_id,
            name=name,
            description="专门负责信息收集、文献调研和数据分析。擅长查找资料、总结信息、识别关键概念。",
            model=model,
            channel=channel
        )
        
        # 研究结果存储
        self.research_results: Dict[str, Any] = {}
    
    def _build_system_prompt(self) -> str:
        """构建系统提示"""
        return """你是一个专业的研究代理。你的职责是：

1. 收集和整理相关信息
2. 分析数据和趋势
3. 识别关键概念和模式
4. 提供准确、全面的研究报告

工作方法：
- 系统性地收集信息
- 从多个角度分析问题
- 验证信息的可靠性
- 清晰地组织和呈现发现

请始终保持客观、准确和全面。"""
    
    def execute_task(self, task: Any) -> Any:
        """
        执行研究任务
        
        参数:
            task: 任务描述，可以是字符串或字典
            
        返回:
            研究结果
        """
        # 解析任务
        if isinstance(task, dict):
            topic = task.get("topic", "")
            depth = task.get("depth", "basic")
            focus_areas = task.get("focus_areas", [])
        else:
            topic = str(task)
            depth = "basic"
            focus_areas = []
        
        print(f"[{self.name}] 开始研究: {topic}")
        
        # 构建研究提示
        research_prompt = f"""请对以下主题进行深入研究：

主题：{topic}

研究深度：{depth}
"""
        
        if focus_areas:
            research_prompt += f"""
重点关注领域：
{', '.join(focus_areas)}
"""
        
        research_prompt += """
请提供：
1. 主题概述
2. 关键概念和定义
3. 主要发现或观点
4. 相关数据或证据
5. 结论和建议

请用清晰、结构化的方式组织你的研究结果。"""
        
        # 调用LLM进行研究
        result = self.call_llm(research_prompt, temperature=0.3, max_tokens=2000)
        
        # 存储结果
        self.research_results[topic] = {
            "topic": topic,
            "depth": depth,
            "result": result
        }
        
        return {
            "topic": topic,
            "research_result": result,
            "status": "completed"
        }
    
    def get_research_summary(self, topic: str) -> str:
        """
        获取研究摘要
        
        参数:
            topic: 研究主题
            
        返回:
            研究摘要
        """
        if topic in self.research_results:
            return self.research_results[topic]["result"]
        return f"未找到关于'{topic}'的研究结果"
```

```python
# agents/coordinator.py
from typing import Any, Dict, List, Optional
from agents.base_agent import BaseAgent

class CoordinatorAgent(BaseAgent):

    """
    
    def __init__(
        self,
        agent_id: str = "coordinator",
        name: str = "协调者",

工作方法：

- 监控进度和质量

请以高效、专业的方式完成协调工作。"""
    
    def register_agent(self, agent: BaseAgent):

        注册代理
        
        参数:
            agent: 代理对象
        """
        self.registered_agents[agent.agent_id] = agent
        print(f"[{self.name}] 注册代理: {agent.name} ({agent.agent_id})")
    
    def execute_task(self, task: Any) -> Any:
        """
        执行协调任务
        
        参数:
            task: 用户任务

        if isinstance(task, dict):
            task_description = task.get("description", "")

        project_id = str(uuid.uuid4())[:8]

        }

        print(f"\n[{self.name}] 分解任务...")

        self.current_project["subtasks"] = subtasks

        print(f"\n[{self.name}] 执行 {len(subtasks)} 个子任务...")
        results = self._execute_subtasks(subtasks)
        
        # 整合结果
        print(f"\n[{self.name}] 整合结果...")
        final_result = self._integrate_results(task_description, results)
        
        self.current_project["status"] = "completed"
        
        return {
            "project_id": project_id,
            "task": task_description,

    def _decompose_task(self, task_description: str, task_type: str) -> List[Dict]:

- 优先级1-5，5最高"""

        # 解析响应
        try:
            import json
            # 提取JSON部分
            json_start = response.find('[')
            json_end = response.rfind(']') + 1
            if json_start != -1 and json_end != -1:
                json_str = response[json_start:json_end]
                subtasks = json.loads(json_str)
            else:
                # 默认子任务
                subtasks = [
                    {
                        "id": "subtask_1",
                        "description": task_description,
                        "assigned_to": list(self.registered_agents.keys())[0] if self.registered_agents else "unknown",
                        "priority": 1,
                        "dependencies": []
                    }
                ]
        except:
            # 默认子任务
            subtasks = [
                {
                    "id": "subtask_1",
                    "description": task_description,
                    "assigned_to": list(self.registered_agents.keys())[0] if self.registered_agents else "unknown",
                    "priority": 1,
                    "dependencies": []
                }
            ]

        执行子任务
        
        参数:
            subtasks: 子任务列表
            
        返回:
            子任务结果字典

            assigned_to = subtask["assigned_to"]

            print(f"  分配给: {assigned_to}")

            for dep_id in dependencies:
                if dep_id not in results:
                    print(f"  等待依赖任务 {dep_id} 完成...")
                    # 这里可以实现更复杂的依赖处理

            if assigned_to in self.registered_agents:
                agent = self.registered_agents[assigned_to]
                
                # 准备任务数据（包含之前的结果）

                    "task": subtask["description"],

                }

                    result = agent.execute_task(task_data)
                    results[subtask_id] = {
                        "subtask": subtask,
                        "result": result,
                        "status": "completed"
                    }
                    print(f"  ✓ 子任务 {subtask_id} 完成")
                except Exception as e:
                    results[subtask_id] = {
                        "subtask": subtask,
                        "error": str(e),
                        "status": "failed"
                    }
                    print(f"  ✗ 子任务 {subtask_id} 失败: {e}")
            else:
                results[subtask_id] = {
                    "subtask": subtask,
                    "error": f"代理 {assigned_to} 未注册",
                    "status": "failed"
                }
                print(f"  ✗ 代理 {assigned_to} 未注册")
        
        return results
    
    def _integrate_results(self, original_task: str, results: Dict[str, Any]) -> str:

子任务结果：
"""
        
        for subtask_id, result_data in results.items():
            status = result_data["status"]
            subtask_desc = result_data["subtask"]["description"]
            
            integrate_prompt += f"\n{subtask_id} ({status}): {subtask_desc}\n"
            
            if status == "completed":
                result = result_data["result"]
                if isinstance(result, dict):
                    for key, value in result.items():
                        if isinstance(value, str) and len(value) > 500:
                            integrate_prompt += f"  {key}: {value[:500]}...\n"
                        else:
                            integrate_prompt += f"  {key}: {value}\n"
                else:
                    result_str = str(result)
                    if len(result_str) > 500:
                        integrate_prompt += f"  结果: {result_str[:500]}...\n"
                    else:
                        integrate_prompt += f"  结果: {result}\n"
            else:
                integrate_prompt += f"  错误: {result_data.get('error', '未知错误')}\n"
        
        integrate_prompt += """
请整合以上结果，生成一个完整、连贯的最终输出。确保：
1. 保留所有重要信息
2. 消除重复内容
3. 组织成清晰的结构
4. 保持一致的风格"""
        
        final_result = self.call_llm(integrate_prompt, temperature=0.5, max_tokens=3000)
        
        return final_result
```

```python
# orchestration/pipeline.py
from typing import Any, Dict, List, Callable
from agents.base_agent import BaseAgent


class PipelineOrchestrator:
    """
    流水线编排器
    代理按顺序执行，前一个的输出是后一个的输入
    """
    
    def __init__(self):
        """初始化流水线编排器"""
        self.pipeline: List[BaseAgent] = []
    
    def add_stage(self, agent: BaseAgent):
        """
        添加流水线阶段
        
        参数:
            agent: 代理对象
        """
        self.pipeline.append(agent)
        print(f"添加流水线阶段: {agent.name}")
    
    def run(self, initial_input: Any) -> Dict[str, Any]:
        """
        运行流水线
        
        参数:
            initial_input: 初始输入
            
        返回:
            执行结果
        """
        print(f"\n{'='*60}")
        print(f"流水线编排器 - 开始执行")
        print(f"阶段数: {len(self.pipeline)}")
        print(f"{'='*60}")
        
        current_input = initial_input
        stage_results = []
        
        for i, agent in enumerate(self.pipeline, 1):
            print(f"\n--- 阶段 {i}: {agent.name} ---")
            
            try:
                # 执行当前阶段
                result = agent.execute_task(current_input)
                
                stage_results.append({
                    "stage": i,
                    "agent": agent.name,
                    "result": result,
                    "status": "completed"
                })
                
                # 将结果传递给下一阶段
                if isinstance(result, dict):
                    # 尝试提取主要内容
                    current_input = result.get("content") or result.get("result") or result
                else:
                    current_input = result
                
                print(f"✓ 阶段 {i} 完成")
                
            except Exception as e:
                stage_results.append({
                    "stage": i,
                    "agent": agent.name,
                    "error": str(e),
                    "status": "failed"
                })
                print(f"✗ 阶段 {i} 失败: {e}")
                break
        
        final_result = current_input
        
        print(f"\n{'='*60}")
        print(f"流水线完成")
        print(f"{'='*60}")
        
        return {
            "initial_input": initial_input,
            "stage_results": stage_results,
            "final_result": final_result,
            "status": "completed" if all(r["status"] == "completed" for r in stage_results) else "failed"
        }
```
