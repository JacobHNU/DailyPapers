## 第一步：理解DPO训练

### 1.1 DPO核心概念

```
DPO（直接偏好优化）
═══════════════════════════════════════════════════════════════════

传统RLHF流程：
1. 收集人类偏好数据
2. 训练奖励模型
3. 使用PPO优化策略

DPO简化流程：
1. 收集偏好对（chosen, rejected）
2. 直接优化策略（跳过奖励模型）

DPO损失函数：
L_DPO = -E[log σ(β · (log π_θ(y_w|x)/π_ref(y_w|x) 
                    - log π_θ(y_l|x)/π_ref(y_l|x)))]

其中：
- y_w：优选响应（winner）
- y_l：劣选响应（loser）
- π_θ：当前策略
- π_ref：参考策略
- β：温度参数

核心洞察：
DPO隐式地训练奖励模型
无需显式的奖励函数

═══════════════════════════════════════════════════════════════════
```

### 1.2 代理轨迹DPO

```
代理轨迹DPO流程
═══════════════════════════════════════════════════════════════════

步骤1：轨迹收集
- 对同一任务执行N个轨迹
- 记录每个轨迹的动作序列
- 评估每个轨迹的成功/失败

步骤2：偏好对构造
- 成功轨迹 = chosen（优选）
- 失败轨迹 = rejected（劣选）
- 可选：按质量排序

步骤3：DPO训练
- 使用偏好对训练策略
- 最大化chosen轨迹的概率
- 最小化rejected轨迹的概率

步骤4：评估
- 使用新策略执行任务
- 比较训练前后的成功率
- 迭代改进

═══════════════════════════════════════════════════════════════════
```

```python
# agents/simple_agent.py
from typing import Any, Dict, List, Optional
from openai import OpenAI
from dataclasses import dataclass
import json


@dataclass
class Action:
    """动作数据类"""
    thought: str
    action: str
    action_input: str
    observation: Optional[str] = None


@dataclass
class Trajectory:
    """轨迹数据类"""
    task: str
    actions: List[Action]
    final_answer: str
    success: bool
    reward: float = 0.0
    metadata: Dict[str, Any] = None
    
    def to_string(self) -> str:
        """转换为字符串表示"""
        parts = [f"Task: {self.task}\n"]
        
        for i, action in enumerate(self.actions, 1):
            parts.append(f"Step {i}:")
            parts.append(f"  Thought: {action.thought}")
            parts.append(f"  Action: {action.action}")
            parts.append(f"  Action Input: {action.action_input}")
            if action.observation:
                parts.append(f"  Observation: {action.observation}")
            parts.append("")
        
        parts.append(f"Final Answer: {self.final_answer}")
        parts.append(f"Success: {self.success}")
        
        return "\n".join(parts)
    
    def to_dict(self) -> Dict:
        """转换为字典"""
        return {
            "task": self.task,
            "actions": [
                {
                    "thought": a.thought,
                    "action": a.action,
                    "action_input": a.action_input,
                    "observation": a.observation
                }
                for a in self.actions
            ],
            "final_answer": self.final_answer,
            "success": self.success,
            "reward": self.reward
        }


class SimpleAgent:
    """
    简单代理
    用于收集训练轨迹
    """
    
    def __init__(
        self,
        model: str = "gpt-3.5-turbo",
        temperature: float = 0.7,
        max_steps: int = 5
    ):
        """
        初始化代理
        
        参数:
            model: 使用的模型
            temperature: 温度参数（控制随机性）
            max_steps: 最大步骤数
        """
        self.client = OpenAI()
        self.model = model
        self.temperature = temperature
        self.max_steps = max_steps
        
        # 定义工具
        self.tools = {
            "search": self._search,
            "calculate": self._calculate,
            "finish": self._finish
        }
    
    def _search(self, query: str) -> str:
        """模拟搜索工具"""
        # 这里可以接入真实搜索API
        return f"搜索结果: 关于'{query}'的信息..."
    
    def _calculate(self, expression: str) -> str:
        """计算工具"""
        try:
            result = eval(expression)
            return f"计算结果: {result}"
        except:
            return "计算错误"
    
    def _finish(self, answer: str) -> str:
        """完成工具"""
        return answer
    
    def run(self, task: str) -> Trajectory:
        """
        执行任务并收集轨迹
        
        参数:
            task: 任务描述
            
        返回:
            轨迹对象
        """
        actions = []
        final_answer = ""
        
        system_prompt = """你是一个有用的助手。你可以使用以下工具：
- search(query): 搜索信息
- calculate(expression): 计算数学表达式
- finish(answer): 提交最终答案

请按以下格式回答：
Thought: [你的思考]
Action: [工具名称]
Action Input: [工具输入]

当你准备好最终答案时：
Thought: 我现在知道答案了
Action: finish
Action Input: [最终答案]"""
        
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": f"Task: {task}"}
        ]
        
        for step in range(self.max_steps):
            # 调用LLM
            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=self.temperature,
                max_tokens=300
            )
            
            output = response.choices[0].message.content
            
            # 解析输出
            thought = self._extract_field(output, "Thought")
            action = self._extract_field(output, "Action")
            action_input = self._extract_field(output, "Action Input")
            
            # 执行动作
            observation = None
            if action in self.tools:
                observation = self.tools[action](action_input)
            
            # 记录动作
            actions.append(Action(
                thought=thought,
                action=action,
                action_input=action_input,
                observation=observation
            ))
            
            # 检查是否完成
            if action == "finish":
                final_answer = action_input
                break
            
            # 更新消息
            messages.append({"role": "assistant", "content": output})
            if observation:
                messages.append({"role": "user", "content": f"Observation: {observation}"})
        
        # 评估成功（这里使用简单规则）
        success = self._evaluate_success(task, final_answer)
        
        return Trajectory(
            task=task,
            actions=actions,
            final_answer=final_answer,
            success=success,
            reward=1.0 if success else 0.0
        )
    
    def _extract_field(self, text: str, field: str) -> str:
        """从文本中提取字段"""
        lines = text.split("\n")
        for line in lines:
            if line.startswith(f"{field}:"):
                return line[len(f"{field}:"):].strip()
        return ""
    
    def _evaluate_success(self, task: str, answer: str) -> bool:
        """
        评估任务成功（简单规则）
        可以替换为更复杂的评估
        """
        # 简单规则：如果答案不为空且不包含"不知道"，认为成功
        if not answer:
            return False
        if "不知道" in answer or "无法" in answer:
            return False
        return True
    
    def collect_trajectories(
        self, 
        task: str, 
        num_trajectories: int = 5
    ) -> List[Trajectory]:
        """
        收集多个轨迹
        
        参数:
            task: 任务描述
            num_trajectories: 轨迹数量
            
        返回:
            轨迹列表
        """
        trajectories = []
        
        for i in range(num_trajectories):
            print(f"收集轨迹 {i+1}/{num_trajectories}...")
            trajectory = self.run(task)
            trajectories.append(trajectory)
            
            # 打印结果
            status = "✓ 成功" if trajectory.success else "✗ 失败"
            print(f"  {status}: {trajectory.final_answer[:50]}...")
        
        return trajectories
```

```python
# collection/collector.py
from typing import List, Dict, Any, Callable
from agents.simple_agent import SimpleAgent, Trajectory
import json
import os


class TrajectoryCollector:
    """
    轨迹收集器
    收集和管理代理轨迹
    """
    
    def __init__(
        self,
        agent: SimpleAgent,
        output_dir: str = "./trajectories"
    ):
        """
        初始化收集器
        
        参数:
            agent: 代理对象
            output_dir: 输出目录
        """
        self.agent = agent
        self.output_dir = output_dir
        
        # 创建输出目录
        os.makedirs(output_dir, exist_ok=True)
    
    def collect_for_task(
        self,
        task: str,
        num_trajectories: int = 5,
        save: bool = True
    ) -> List[Trajectory]:
        """
        为单个任务收集轨迹
        
        参数:
            task: 任务描述
            num_trajectories: 轨迹数量
            save: 是否保存
            
        返回:
            轨迹列表
        """
        print(f"\n收集轨迹: {task}")
        print(f"目标数量: {num_trajectories}")
        
        trajectories = self.agent.collect_trajectories(task, num_trajectories)
        
        # 统计
        success_count = sum(1 for t in trajectories if t.success)
        print(f"\n统计: {success_count}/{num_trajectories} 成功")
        
        # 保存
        if save:
            self._save_trajectories(trajectories, task)
        
        return trajectories
    
    def collect_for_tasks(
        self,
        tasks: List[str],
        num_trajectories_per_task: int = 5
    ) -> Dict[str, List[Trajectory]]:
        """
        为多个任务收集轨迹
        
        参数:
            tasks: 任务列表
            num_trajectories_per_task: 每个任务的轨迹数量
            
        返回:
            任务 -> 轨迹列表 的字典
        """
        all_trajectories = {}
        
        for i, task in enumerate(tasks, 1):
            print(f"\n{'='*60}")
            print(f"任务 {i}/{len(tasks)}")
            print(f"{'='*60}")
            
            trajectories = self.collect_for_task(task, num_trajectories_per_task)
            all_trajectories[task] = trajectories
        
        return all_trajectories
    
    def _save_trajectories(self, trajectories: List[Trajectory], task: str):
        """保存轨迹到文件"""
        # 生成文件名
        task_hash = hash(task) % 10000
        filename = f"trajectories_{task_hash}.json"
        filepath = os.path.join(self.output_dir, filename)
        
        # 转换为可序列化格式
        data = {
            "task": task,
            "trajectories": [t.to_dict() for t in trajectories]
        }
        
        # 保存
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        
        print(f"轨迹已保存到: {filepath}")
    
    def load_trajectories(self, filepath: str) -> List[Trajectory]:
        """
        从文件加载轨迹
        
        参数:
            filepath: 文件路径
            
        返回:
            轨迹列表
        """
        with open(filepath, 'r', encoding='utf-8') as f:
            data = json.load(f)
        
        trajectories = []
        for t_data in data["trajectories"]:
            from agents.simple_agent import Action
            actions = [
                Action(
                    thought=a["thought"],
                    action=a["action"],
                    action_input=a["action_input"],
                    observation=a.get("observation")
                )
                for a in t_data["actions"]
            ]
            
            trajectories.append(Trajectory(
                task=t_data["task"],
                actions=actions,
                final_answer=t_data["final_answer"],
                success=t_data["success"],
                reward=t_data.get("reward", 0)
            ))
        
        return trajectories
```

```python
# training/dataset.py
from typing import List, Dict, Tuple
from agents.simple_agent import Trajectory
from dataclasses import dataclass
import random


@dataclass
class PreferencePair:
    """偏好对数据类"""
    task: str
    chosen: Trajectory  # 优选轨迹
    rejected: Trajectory  # 劣选轨迹
    
    def to_dict(self) -> Dict:
        """转换为字典"""
        return {
            "task": self.task,
            "chosen": self.chosen.to_dict(),
            "rejected": self.rejected.to_dict()
        }


class PreferenceDatasetBuilder:
    """
    偏好数据集构建器
    从轨迹中构建偏好对
    """
    
    def __init__(self):
        """初始化数据集构建器"""
        pass
    
    def build_from_trajectories(
        self,
        trajectories: List[Trajectory]
    ) -> List[PreferencePair]:
        """
        从轨迹列表构建偏好对
        
        参数:
            trajectories: 轨迹列表
            
        返回:
            偏好对列表
        """
        # 分离成功和失败轨迹
        successful = [t for t in trajectories if t.success]
        failed = [t for t in trajectories if not t.success]
        
        preference_pairs = []
        
        # 配对：成功 vs 失败
        for success_traj in successful:
            for fail_traj in failed:
                pair = PreferencePair(
                    task=success_traj.task,
                    chosen=success_traj,
                    rejected=fail_traj
                )
                preference_pairs.append(pair)
        
        return preference_pairs
    
    def build_from_multiple_tasks(
        self,
        all_trajectories: Dict[str, List[Trajectory]]
    ) -> List[PreferencePair]:
        """
        从多个任务的轨迹构建偏好对
        
        参数:
            all_trajectories: 任务 -> 轨迹列表 的字典
            
        返回:
            偏好对列表
        """
        all_pairs = []
        
        for task, trajectories in all_trajectories.items():
            pairs = self.build_from_trajectories(trajectories)
            all_pairs.extend(pairs)
        
        return all_pairs
    
    def build_with_ranking(
        self,
        trajectories: List[Trajectory]
    ) -> List[PreferencePair]:
        """
        基于排名构建偏好对
        
        参数:
            trajectories: 轨迹列表（按质量排序）
            
        返回:
            偏好对列表
        """
        # 按奖励排序
        sorted_trajectories = sorted(
            trajectories,
            key=lambda t: t.reward,
            reverse=True
        )
        
        preference_pairs = []
        
        # 配对：高奖励 vs 低奖励
        for i in range(len(sorted_trajectories)):
            for j in range(i + 1, len(sorted_trajectories)):
                pair = PreferencePair(
                    task=sorted_trajectories[i].task,
                    chosen=sorted_trajectories[i],
                    rejected=sorted_trajectories[j]
                )
                preference_pairs.append(pair)
        
        return preference_pairs
    
    def format_for_dpo(
        self,
        preference_pairs: List[PreferencePair]
    ) -> List[Dict[str, str]]:
        """
        格式化为DPO训练格式
        
        参数:
            preference_pairs: 偏好对列表
            
        返回:
            DPO格式的数据列表
        """
        dpo_data = []
        
        for pair in preference_pairs:
            # 构建提示
            prompt = f"Task: {pair.task}\n\nPlease complete this task step by step."
            
            # 构建chosen响应
            chosen_response = self._trajectory_to_response(pair.chosen)
            
            # 构建rejected响应
            rejected_response = self._trajectory_to_response(pair.rejected)
            
            dpo_data.append({
                "prompt": prompt,
                "chosen": chosen_response,
                "rejected": rejected_response
            })
        
        return dpo_data
    
    def _trajectory_to_response(self, trajectory: Trajectory) -> str:
        """将轨迹转换为响应字符串"""
        parts = []
        
        for i, action in enumerate(trajectory.actions, 1):
            parts.append(f"Thought: {action.thought}")
            parts.append(f"Action: {action.action}")
            parts.append(f"Action Input: {action.action_input}")
            if action.observation:
                parts.append(f"Observation: {action.observation}")
            parts.append("")
        
        parts.append(f"Final Answer: {trajectory.final_answer}")
        
        return "\n".join(parts)
```

```python
from typing import List, Dict, Optional

        self,
        model_name: str = "gpt2",  # 使用小模型进行演示
        learning_rate: float = 5e-5,
        batch_size: int = 4,
        max_epochs: int = 3,
        beta: float = 0.1
    ):
        """
        初始化训练器
        
        参数:
            model_name: 基础模型名称
            learning_rate: 学习率
            batch_size: 批次大小
            max_epochs: 最大训练轮数
            beta: DPO beta参数
        """
        self.model_name = model_name
        self.learning_rate = learning_rate
        self.batch_size = batch_size
        self.max_epochs = max_epochs
        self.beta = beta
        
        # 加载模型和分词器
        print(f"加载模型: {model_name}")
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        
        # 设置pad token
        if self.tokenizer.pad_token is None:
            self.tokenizer.pad_token = self.tokenizer.eos_token
    
    def prepare_dataset(
        self,
        preference_data: List[Dict[str, str]]
    ) -> Dataset:
        """
        准备训练数据集
        
        参数:
            preference_data: DPO格式的数据列表
            
        返回:
            HuggingFace Dataset对象
        """
        # 转换为Dataset格式
        dataset = Dataset.from_list(preference_data)
        
        print(f"数据集大小: {len(dataset)}")
        print(f"数据列: {dataset.column_names}")
        
        return dataset
    
    def train(
        self,
        dataset: Dataset,
        output_dir: str = "./dpo_model"
    ):
        """
        执行DPO训练
        
        参数:
            dataset: 训练数据集
            output_dir: 输出目录
        """
        print("\n开始DPO训练...")
        
        # 配置训练参数
        training_args = DPOConfig(
            output_dir=output_dir,
            learning_rate=self.learning_rate,
            per_device_train_batch_size=self.batch_size,
            num_train_epochs=self.max_epochs,
            beta=self.beta,
            logging_steps=10,
            save_strategy="epoch",
            remove_unused_columns=False,
            gradient_accumulation_steps=2,
            warmup_steps=100,
        )
        
        # 创建训练器
        trainer = DPOTrainer(
            model=self.model,
            args=training_args,
            train_dataset=dataset,
            processing_class=self.tokenizer,
        )
        
        # 执行训练
        trainer.train()
        
        # 保存模型
        trainer.save_model(output_dir)
        self.tokenizer.save_pretrained(output_dir)
        
        print(f"\n模型已保存到: {output_dir}")
    
    def train_from_preference_pairs(
        self,
        preference_pairs: List[Dict],
        output_dir: str = "./dpo_model"
    ):
        """
        从偏好对直接训练
        
        参数:
            preference_pairs: 偏好对列表
            output_dir: 输出目录
        """
        # 准备数据集
        dataset = self.prepare_dataset(preference_pairs)
        
        # 训练
        self.train(dataset, output_dir)
```

```python
# training/simple_dpo.py
from typing import List, Dict
from agents.simple_agent import Trajectory, SimpleAgent
import json


class SimpleDPOTrainer:
    """
    简化的DPO训练器
    使用提示工程模拟DPO效果
    适用于API模型（无法微调）
    """
    
    def __init__(self, agent: SimpleAgent):
        """
        初始化训练器
        
        参数:
            agent: 代理对象
        """
        self.agent = agent
        self.learned_patterns: List[str] = []
    
    def extract_patterns(
        self,
        successful_trajectories: List[Trajectory],
        failed_trajectories: List[Trajectory]
    ) -> List[str]:
        """
        从成功和失败轨迹中提取模式
        
        参数:
            successful_trajectories: 成功轨迹列表
            failed_trajectories: 失败轨迹列表
            
        返回:
            学习到的模式列表
        """
        patterns = []
        
        # 分析成功轨迹的共同特征
        if successful_trajectories:
            success_prompt = f"""分析以下成功轨迹，提取共同的成功模式：

成功轨迹1：
{successful_trajectories[0].to_string()}

成功轨迹2：
{successful_trajectories[1].to_string() if len(successful_trajectories) > 1 else 'N/A'}

请列出3-5个关键的成功模式。"""
            
            response = self.agent.client.chat.completions.create(
                model=self.agent.model,
                messages=[{"role": "user", "content": success_prompt}],
                temperature=0.3,
                max_tokens=500
            )
            
            success_patterns = response.choices[0].message.content
            patterns.append("成功模式:\n" + success_patterns)
        
        # 分析失败轨迹的共同问题
        if failed_trajectories:
            fail_prompt = f"""分析以下失败轨迹，提取常见的失败原因：

失败轨迹1：
{failed_trajectories[0].to_string()}

失败轨迹2：
{failed_trajectories[1].to_string() if len(failed_trajectories) > 1 else 'N/A'}

请列出3-5个常见的失败原因和改进建议。"""
            
            response = self.agent.client.chat.completions.create(
                model=self.agent.model,
                messages=[{"role": "user", "content": fail_prompt}],
                temperature=0.3,
                max_tokens=500
            )
            
            fail_patterns = response.choices[0].message.content
            patterns.append("失败模式:\n" + fail_patterns)
        
        self.learned_patterns = patterns
        return patterns
    
    def create_improved_prompt(self, task: str) -> str:
        """
        创建改进的提示
        
        参数:
            task: 任务描述
            
        返回:
            改进后的提示
        """
        base_prompt = f"""Task: {task}

"""
        
        if self.learned_patterns:
            patterns_text = "\n\n".join(self.learned_patterns)
            base_prompt += f"""根据之前的经验，请注意以下模式：

{patterns_text}

请基于这些经验，更好地完成任务。

"""
        
        base_prompt += """请按以下格式回答：
Thought: [你的思考]
Action: [工具名称]
Action Input: [工具输入]

当你准备好最终答案时：
Thought: 我现在知道答案了
Action: finish
Action Input: [最终答案]"""
        
        return base_prompt
    
    def run_improved(self, task: str) -> Trajectory:
        """
        使用改进的提示运行任务
        
        参数:
            task: 任务描述
            
        返回:
            轨迹对象
        """
        # 创建改进的提示
        improved_prompt = self.create_improved_prompt(task)
        
        # 临时修改代理的系统提示
        original_model = self.agent.model
        
        # 使用较低温度（更确定性）
        original_temp = self.agent.temperature
        self.agent.temperature = 0.3
        
        # 执行任务
        trajectory = self.agent.run(task)
        
        # 恢复温度
        self.agent.temperature = original_temp
        
        return trajectory
    
    def iterative_improvement(
        self,
        task: str,
        num_iterations: int = 3,
        trajectories_per_iteration: int = 5
    ) -> List[Trajectory]:
        """
        迭代改进
        
        参数:
            task: 任务描述
            num_iterations: 迭代次数
            trajectories_per_iteration: 每次迭代的轨迹数
            
        返回:
            最终轨迹列表
        """
        print(f"\n开始迭代改进: {task}")
        print(f"迭代次数: {num_iterations}")
        
        all_trajectories = []
        
        for iteration in range(num_iterations):
            print(f"\n--- 迭代 {iteration + 1} ---")
            
            # 收集轨迹
            trajectories = self.agent.collect_trajectories(
                task, 
                trajectories_per_iteration
            )
            
            all_trajectories.extend(trajectories)
            
            # 分析轨迹
            successful = [t for t in trajectories if t.success]
            failed = [t for t in trajectories if not t.success]
            
            print(f"成功: {len(successful)}, 失败: {len(failed)}")
            
            # 提取模式
            if successful and failed:
                patterns = self.extract_patterns(successful, failed)
                print(f"学习到 {len(patterns)} 个模式")
            
            # 使用改进的提示运行
            if iteration < num_iterations - 1:
                print("\n使用改进的提示测试...")
                improved_traj = self.run_improved(task)
                print(f"改进结果: {'成功' if improved_traj.success else '失败'}")
        
        return all_trajectories
```

```python
# evaluation/evaluator.py
from typing import List, Dict
from agents.simple_agent import SimpleAgent, Trajectory
import statistics


class AgentEvaluator:
    """
    代理评估器
    评估代理性能
    """
    
    def __init__(self, agent: SimpleAgent):
        """
        初始化评估器
        
        参数:
            agent: 代理对象
        """
        self.agent = agent
    
    def evaluate_on_tasks(
        self,
        tasks: List[str],
        num_trials: int = 5
    ) -> Dict:
        """
        在多个任务上评估
        
        参数:
            tasks: 任务列表
            num_trials: 每个任务的试验次数
            
        返回:
            评估结果
        """
        results = {}
        
        for task in tasks:
            print(f"\n评估任务: {task[:50]}...")
            
            trajectories = self.agent.collect_trajectories(task, num_trials)
            
            success_count = sum(1 for t in trajectories if t.success)
            success_rate = success_count / num_trials
            
            avg_reward = statistics.mean([t.reward for t in trajectories])
            avg_steps = statistics.mean([len(t.actions) for t in trajectories])
            
            results[task] = {
                "success_rate": success_rate,
                "avg_reward": avg_reward,
                "avg_steps": avg_steps,
                "trajectories": trajectories
            }
            
            print(f"  成功率: {success_rate:.2%}")
            print(f"  平均奖励: {avg_reward:.2f}")
            print(f"  平均步骤: {avg_steps:.1f}")
        
        # 计算总体统计
        all_success_rates = [r["success_rate"] for r in results.values()]
        
        overall = {
            "avg_success_rate": statistics.mean(all_success_rates),
            "min_success_rate": min(all_success_rates),
            "max_success_rate": max(all_success_rates),
            "num_tasks": len(tasks)
        }
        
        return {
            "per_task": results,
            "overall": overall
        }
    
    def compare_agents(
        self,
        agent1: SimpleAgent,
        agent2: SimpleAgent,
        tasks: List[str],
        num_trials: int = 5
    ) -> Dict:
        """
        比较两个代理
        
        参数:
            agent1: 代理1
            agent2: 代理2
            tasks: 任务列表
            num_trials: 试验次数
            
        返回:
            比较结果
        """
        print("\n评估代理1...")
        self.agent = agent1
        results1 = self.evaluate_on_tasks(tasks, num_trials)
        
        print("\n评估代理2...")
        self.agent = agent2
        results2 = self.evaluate_on_tasks(tasks, num_trials)
        
        # 比较
        comparison = {
            "agent1_results": results1,
            "agent2_results": results2,
            "improvement": {
                "success_rate": results2["overall"]["avg_success_rate"] - results1["overall"]["avg_success_rate"],
                "reward": 0  # 可以添加更多指标
            }
        }
        
        print(f"\n比较结果:")
        print(f"  代理1 平均成功率: {results1['overall']['avg_success_rate']:.2%}")
        print(f"  代理2 平均成功率: {results2['overall']['avg_success_rate']:.2%}")
        print(f"  提升: {comparison['improvement']['success_rate']:.2%}")
        
        return comparison
```
